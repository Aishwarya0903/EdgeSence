cmake_minimum_required(VERSION 3.20)
project(edgesense_firmware LANGUAGES CXX)

option(EDGESENSE_BUILD_TESTS "Build GoogleTest protocol tests" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_library(edgesense_core src/protocol.cpp src/sensor.cpp)
target_include_directories(edgesense_core PUBLIC include)

add_executable(edgesense_firmware src/main.cpp)
target_link_libraries(edgesense_firmware PRIVATE edgesense_core)
if(WIN32)
  target_link_libraries(edgesense_firmware PRIVATE ws2_32)
endif()

if(EDGESENSE_BUILD_TESTS)
  include(CTest)
  find_package(GTest QUIET)
  if(NOT GTest_FOUND)
    include(FetchContent)
    FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.14.0.zip)
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(googletest)
  endif()
  add_executable(protocol_tests tests/protocol_test.cpp)
  target_link_libraries(protocol_tests PRIVATE edgesense_core GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(protocol_tests)
endif()
